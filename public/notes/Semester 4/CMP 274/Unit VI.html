6.1 Gauss Elimination with Partial Pivoting

Converts the coefficient matrix into an upper triangular form, followed by back-substitution. Pivoting swaps rows to place the largest coefficient in the pivot position before performing row operations, minimizing round-off error.

Gauss Elimination C Implementation:


#include <stdio.h>
#include <math.h>

#define N 3

void gauss_elimination(double a[N][N+1]) {
    // Step 1: Forward Elimination with Partial Pivoting
    for (int i = 0; i < N-1; i++) {
        // Pivot check and row swap
        int max_row = i;
        for (int k = i+1; k < N; k++) {
            if (fabs(a[k][i]) > fabs(a[max_row][i])) {
                max_row = k;
            }
        }
        
        // Swap rows
        for (int col = 0; col <= N; col++) {
            double temp = a[i][col];
            a[i][col] = a[max_row][col];
            a[max_row][col] = temp;
        }
        
        // Eliminate column elements
        for (int k = i+1; k < N; k++) {
            double factor = a[k][i] / a[i][i];
            for (int j = i; j <= N; j++) {
                a[k][j] -= factor * a[i][j];
            }
        }
    }
    
    // Step 2: Back-Substitution
    double x[N];
    for (int i = N - 1; i >= 0; i--) {
        x[i] = a[i][N];
        for (int j = i + 1; j < N; j++) {
            x[i] -= a[i][j] * x[j];
        }
        x[i] /= a[i][i];
    }
    
    printf("\nSolution vector by Gauss Elimination:\n");
    for (int i = 0; i < N; i++) {
        printf("x[%d] = %lf\n", i, x[i]);
    }
}

int main() {
    double a[N][N+1] = {
        {2.0, 1.0, 1.0, 10.0},
        {1.0, 4.0, 2.0, 21.0},
        {1.0, 2.0, 5.0, 30.0}
    };
    gauss_elimination(a);
    return 0;
}

6.2 Gauss-Jordan Method

Converts the coefficient matrix directly into a diagonal identity matrix using row operations, eliminating the need for back-substitution.

Gauss-Jordan C Implementation:


#include <stdio.h>
#include <math.h>

#define N 3

void gauss_jordan(double a[N][N+1]) {
    for (int i = 0; i < N; i++) {
        // Pivot selection
        double pivot = a[i][i];
        
        // Normalize pivot row
        for (int j = 0; j <= N; j++) {
            a[i][j] /= pivot;
        }
        
        // Eliminate other column entries
        for (int k = 0; k < N; k++) {
            if (k != i) {
                double factor = a[k][i];
                for (int j = 0; j <= N; j++) {
                    a[k][j] -= factor * a[i][j];
                }
            }
        }
    }
    
    printf("\nSolution vector by Gauss-Jordan:\n");
    for (int i = 0; i < N; i++) {
        printf("x[%d] = %lf\n", i, a[i][N]);
    }
}

int main() {
    double a[N][N+1] = {
        {2.0, 1.0, 1.0, 10.0},
        {1.0, 4.0, 2.0, 21.0},
        {1.0, 2.0, 5.0, 30.0}
    };
    gauss_jordan(a);
    return 0;
}

6.3 LU Decomposition (Doolittle Method)

Decomposes coefficient matrix A into Lower triangular L (with diagonal elements 1) and Upper triangular U such that A = L * U.

LU Decomposition C Implementation:


#include <stdio.h>

#define N 3

void lu_decomposition(double a[N][N], double b[N]) {
    double L[N][N] = {0};
    double U[N][N] = {0};
    
    for (int i = 0; i < N; i++) {
        L[i][i] = 1.0;
    }
    
    for (int i = 0; i < N; i++) {
        // Upper Triangular
        for (int k = i; k < N; k++) {
            double sum = 0.0;
            for (int j = 0; j < i; j++) {
                sum += L[i][j] * U[j][k];
            }
            U[i][k] = a[i][k] - sum;
        }
        
        // Lower Triangular
        for (int k = i + 1; k < N; k++) {
            double sum = 0.0;
            for (int j = 0; j < i; j++) {
                sum += L[k][j] * U[j][i];
            }
            L[k][i] = (a[k][i] - sum) / U[i][i];
        }
    }
    
    // Solve L*y = b (Forward Substitution)
    double y[N];
    for (int i = 0; i < N; i++) {
        y[i] = b[i];
        for (int j = 0; j < i; j++) {
            y[i] -= L[i][j] * y[j];
        }
    }
    
    // Solve U*x = y (Back-Substitution)
    double x[N];
    for (int i = N - 1; i >= 0; i--) {
        x[i] = y[i];
        for (int j = i + 1; j < N; j++) {
            x[i] -= U[i][j] * x[j];
        }
        x[i] /= U[i][i];
    }
    
    printf("\nSolution vector by LU Decomposition:\n");
    for (int i = 0; i < N; i++) {
        printf("x[%d] = %lf\n", i, x[i]);
    }
}

int main() {
    double a[N][N] = {
        {2.0, 1.0, 1.0},
        {1.0, 4.0, 2.0},
        {1.0, 2.0, 5.0}
    };
    double b[N] = {10.0, 21.0, 30.0};
    lu_decomposition(a, b);
    return 0;
}

6.4 Gauss-Seidel Iterative Method

An iterative method that updates variables step-by-step. It uses the newly calculated values of variables immediately in the current iteration step to speed up convergence.

Gauss-Seidel C Implementation:


#include <stdio.h>
#include <math.h>

#define N 3

void gauss_seidel(double a[N][N], double b[N], double tol) {
    double x[N] = {0.0}; // Initial guesses
    double prev_x[N];
    int max_iterations = 100;
    
    printf("\nGauss-Seidel Iterations:\n");
    printf("%-10s %-12s %-12s %-12s\n", "Iter", "x[0]", "x[1]", "x[2]");
    
    for (int iter = 1; iter <= max_iterations; iter++) {
        for (int i = 0; i < N; i++) {
            prev_x[i] = x[i];
        }
        
        for (int i = 0; i < N; i++) {
            double sum = b[i];
            for (int j = 0; j < N; j++) {
                if (j != i) {
                    sum -= a[i][j] * x[j];
                }
            }
            x[i] = sum / a[i][i];
        }
        
        printf("%-10d %-12.6f %-12.6f %-12.6f\n", iter, x[0], x[1], x[2]);
        
        // Convergence check
        double max_err = 0.0;
        for (int i = 0; i < N; i++) {
            double err = fabs(x[i] - prev_x[i]);
            if (err > max_err) max_err = err;
        }
        
        if (max_err < tol) {
            printf("\nConverged in %d iterations.\n", iter);
            break;
        }
    }
}

int main() {
    double a[N][N] = {
        {5.0, -1.0, 1.0},
        {1.0,  6.0, 2.0},
        {1.0,  2.0, 7.0}
    };
    double b[N] = {10.0, 21.0, 30.0};
    gauss_seidel(a, b, 0.0001);
    return 0;
}

6.5 The Power Method (Eigenvalues & Eigenvectors)

An iterative procedure used to find the dominant eigenvalue (largest in absolute value) and its corresponding eigenvector.

Power Method C Implementation:


#include <stdio.h>
#include <math.h>

#define N 3

void power_method(double a[N][N], double tol) {
    double x[N] = {1.0, 1.0, 1.0}; // Initial eigenvector guess
    double y[N];
    double lambda_old = 0.0;
    double lambda_new = 0.0;
    int max_iter = 100;
    
    printf("\nPower Method Iterations:\n");
    for (int step = 1; step <= max_iter; step++) {
        // Compute y = A * x
        for (int i = 0; i < N; i++) {
            y[i] = 0.0;
            for (int j = 0; j < N; j++) {
                y[i] += a[i][j] * x[j];
            }
        }
        
        // Find dominant scale factor
        lambda_new = y[0];
        for (int i = 1; i < N; i++) {
            if (fabs(y[i]) > fabs(lambda_new)) {
                lambda_new = y[i];
            }
        }
        
        // Normalize vector
        for (int i = 0; i < N; i++) {
            x[i] = y[i] / lambda_new;
        }
        
        printf("Iter %2d: Eigenvalue = %lf, Vector = [%.3f, %.3f, %.3f]^T\n", 
               step, lambda_new, x[0], x[1], x[2]);
               
        if (fabs(lambda_new - lambda_old) < tol) {
            printf("\nPower Method converged to Dominant Eigenvalue: %lf\n", lambda_new);
            break;
        }
        lambda_old = lambda_new;
    }
}

int main() {
    double a[N][N] = {
        {4.0, 1.0, 0.0},
        {1.0, 3.0, 1.0},
        {0.0, 1.0, 2.0}
    };
    power_method(a, 0.0001);
    return 0;
}
