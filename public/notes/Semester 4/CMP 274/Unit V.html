5.1 Euler's Method

Euler's Method is a first-order numerical procedure for solving Initial Value Problems (IVPs). It projects the next value along the tangent line slope at the current point.

Derivation using Taylor Series:

The Taylor series expansion of the solution function y(x) around x_i is:
y(x_i + h) = y(x_i) + h * y'(x_i) + (h^2 / 2) * y''(c)

Since y'(x) = f(x, y), substituting this and truncating the terms of order O(h^2) yields the Euler formula:
y_i+1 = y_i + h * f(x_i, y_i)

Euler's Method C Implementation:


#include <stdio.h>

// Define function dy/dx = f(x, y) = x + y
double f(double x, double y) {
    return x + y;
}

void euler(double x0, double y0, double h, double target_x) {
    double x = x0;
    double y = y0;
    printf("%-10s %-15s %-15s\n", "Step", "x", "y");
    
    int step = 0;
    printf("%-10d %-15.6f %-15.6f\n", step++, x, y);
    
    while (x < target_x - 1e-9) {
        y = y + h * f(x, y);
        x = x + h;
        printf("%-10d %-15.6f %-15.6f\n", step++, x, y);
    }
}

int main() {
    euler(0.0, 1.0, 0.1, 0.5);
    return 0;
}

5.2 Heun's Method (Modified Euler Method)

Heun's Method improves accuracy by predicting the next value using Euler's formula, calculating the slope at the predicted point, and averaging it with the initial slope to obtain a corrected value.

Heun's Method C Implementation:


#include <stdio.h>

double f(double x, double y) {
    return x + y;
}

void heun(double x0, double y0, double h, double target_x) {
    double x = x0;
    double y = y0;
    printf("%-10s %-15s %-15s %-15s\n", "Step", "x", "y (Predicted)", "y (Corrected)");
    
    int step = 0;
    printf("%-10d %-15.6f %-15s %-15.6f\n", step++, x, "-", y);
    
    while (x < target_x - 1e-9) {
        double slope1 = f(x, y);
        double y_pred = y + h * slope1; // Predictor step
        double slope2 = f(x + h, y_pred);
        y = y + (h / 2.0) * (slope1 + slope2); // Corrector step
        x = x + h;
        printf("%-10d %-15.6f %-15.6f %-15.6f\n", step++, x, y_pred, y);
    }
}

int main() {
    heun(0.0, 1.0, 0.1, 0.5);
    return 0;
}

5.3 Fourth-Order Runge-Kutta Method (RK-4)

Calculates the slope at the beginning, midpoint, and end of the step interval to compute a weighted average slope.

RK-4 Method C Implementation:


#include <stdio.h>

double f(double x, double y) {
    return x + y;
}

double rk4_step(double x, double y, double h) {
    double k1 = h * f(x, y);
    double k2 = h * f(x + h/2.0, y + k1/2.0);
    double k3 = h * f(x + h/2.0, y + k2/2.0);
    double k4 = h * f(x + h, y + k3);
    return y + (1.0/6.0) * (k1 + 2.0*k2 + 2.0*k3 + k4);
}

void rk4(double x0, double y0, double h, double target_x) {
    double x = x0;
    double y = y0;
    printf("%-10s %-15s %-15s\n", "Step", "x", "y (RK4)");
    
    int step = 0;
    printf("%-10d %-15.6f %-15.6f\n", step++, x, y);
    
    while (x < target_x - 1e-9) {
        y = rk4_step(x, y, h);
        x = x + h;
        printf("%-10d %-15.6f %-15.6f\n", step++, x, y);
    }
}

int main() {
    rk4(0.0, 1.0, 0.1, 0.5);
    return 0;
}

5.4 Solution of Higher-Order Differential Equations

Numerical methods solve systems of first-order differential equations. To solve a higher-order ODE, we must first convert it into a system of first-order equations.

Higher-Order ODE Solver C Implementation:

The following C program solves the second-order ODE: y'' - 3y' + 2y = e^x (equivalent to system: dy/dx = z, dz/dx = 3z - 2y + e^x) using the RK-4 method:


#include <stdio.h>
#include <math.h>

// dy/dx = z
double dy_dx(double x, double y, double z) {
    return z;
}

// dz/dx = 3z - 2y + e^x
double dz_dx(double x, double y, double z) {
    return 3.0*z - 2.0*y + exp(x);
}

void rk4_system(double x0, double y0, double z0, double h, double target_x) {
    double x = x0;
    double y = y0;
    double z = z0;
    
    printf("%-10s %-15s %-15s %-15s\n", "Step", "x", "y", "y' (z)");
    int step = 0;
    printf("%-10d %-15.6f %-15.6f %-15.6f\n", step++, x, y, z);
    
    while (x < target_x - 1e-9) {
        double ky1 = h * dy_dx(x, y, z);
        double kz1 = h * dz_dx(x, y, z);
        
        double ky2 = h * dy_dx(x + h/2.0, y + ky1/2.0, z + kz1/2.0);
        double kz2 = h * dz_dx(x + h/2.0, y + ky1/2.0, z + kz1/2.0);
        
        double ky3 = h * dy_dx(x + h/2.0, y + ky2/2.0, z + kz2/2.0);
        double kz3 = h * dz_dx(x + h/2.0, y + ky2/2.0, z + kz2/2.0);
        
        double ky4 = h * dy_dx(x + h, y + ky3, z + kz3);
        double kz4 = h * dz_dx(x + h, y + ky3, z + kz3);
        
        y = y + (1.0/6.0) * (ky1 + 2.0*ky2 + 2.0*ky3 + ky4);
        z = z + (1.0/6.0) * (kz1 + 2.0*kz2 + 2.0*kz3 + kz4);
        x = x + h;
        
        printf("%-10d %-15.6f %-15.6f %-15.6f\n", step++, x, y, z);
    }
}

int main() {
    rk4_system(0.0, 1.0, 0.0, 0.1, 0.5);
    return 0;
}

5.5 Boundary Value Problems (Shooting Method)

Converts a BVP into an equivalent IVP by guessing the missing initial conditions and refining them iteratively.

Shooting Method C Implementation:


#include <stdio.h>
#include <math.h>

// System equations representing y'' = f(x, y, y')
double dy(double x, double y, double z) { return z; }
double dz(double x, double y, double z) { return 3.0*z - 2.0*y + exp(x); }

// Helper: Run RK4 system integration to calculate boundary y(b)
double integrate_system(double x0, double y0, double initial_slope, double h, double b) {
    double x = x0;
    double y = y0;
    double z = initial_slope;
    
    while (x < b - 1e-9) {
        double ky1 = h * dy(x, y, z);
        double kz1 = h * dz(x, y, z);
        
        double ky2 = h * dy(x + h/2.0, y + ky1/2.0, z + kz1/2.0);
        double kz2 = h * dz(x + h/2.0, y + ky1/2.0, z + kz1/2.0);
        
        double ky3 = h * dy(x + h/2.0, y + ky2/2.0, z + kz2/2.0);
        double kz3 = h * dz(x + h/2.0, y + ky2/2.0, z + kz2/2.0);
        
        double ky4 = h * dy(x + h, y + ky3, z + kz3);
        double kz4 = h * dz(x + h, y + ky3, z + kz3);
        
        y = y + (1.0/6.0) * (ky1 + 2.0*ky2 + 2.0*ky3 + ky4);
        z = z + (1.0/6.0) * (kz1 + 2.0*kz2 + 2.0*kz3 + kz4);
        x = x + h;
    }
    return y;
}

void shooting_method(double x0, double y0, double b, double target_yb, double h) {
    // We guess two initial slopes and interpolate using linear secant line
    double m1 = 0.0;
    double yb1 = integrate_system(x0, y0, m1, h, b);
    
    double m2 = 1.0;
    double yb2 = integrate_system(x0, y0, m2, h, b);
    
    double m3 = m2 - (yb2 - target_yb) * (m2 - m1) / (yb2 - yb1); // Secant correction
    double yb3 = integrate_system(x0, y0, m3, h, b);
    
    printf("Shooting Method Results:\n");
    printf("Guess 1 (slope: %lf) yielded y(b) = %lf\n", m1, yb1);
    printf("Guess 2 (slope: %lf) yielded y(b) = %lf\n", m2, yb2);
    printf("Corrected slope: %lf yielded y(b) = %lf (Target: %lf)\n", m3, yb3, target_yb);
}

int main() {
    shooting_method(0.0, 1.0, 0.5, 2.5, 0.1);
    return 0;
}
