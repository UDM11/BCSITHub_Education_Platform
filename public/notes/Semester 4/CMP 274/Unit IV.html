4.1 Numerical Differentiation

Numerical differentiation calculates the derivative of a function at a point using values of the function. It is derived using Taylor series expansions or difference operators.

Central Difference First-Derivative Derivation:

Using Taylor expansions around x with step h:


1) f(x + h) = f(x) + h*f'(x) + (h^2/2)*f''(x) + (h^3/6)*f'''(x) + ...
2) f(x - h) = f(x) - h*f'(x) + (h^2/2)*f''(x) - (h^3/6)*f'''(x) + ...

Subtracting equation 2 from equation 1:
f(x + h) - f(x - h) = 2*h*f'(x) + (h^3/3)*f'''(x) + ...

Solving for f'(x):
f'(x) = [ f(x + h) - f(x - h) ] / (2*h)  -  (h^2/6)*f'''(c)

This shows the Central Difference approximation has an error term of order O(h^2).

Numerical Differentiation C Implementation:


#include <stdio.h>
#include <math.h>

// Function f(x) = sin(x)
double f(double x) {
    return sin(x);
}

int main() {
    double x = 1.0; // Point to evaluate derivative
    double h = 0.1; // Step size
    
    double forward_diff = (f(x + h) - f(x)) / h;
    double backward_diff = (f(x) - f(x - h)) / h;
    double central_diff = (f(x + h) - f(x - h)) / (2.0 * h);
    double exact_val = cos(x); // True derivative of sin(x) is cos(x)
    
    printf("Exact derivative: %lf\n", exact_val);
    printf("Forward Difference: %lf (Err: %e)\n", forward_diff, fabs(exact_val - forward_diff));
    printf("Backward Difference: %lf (Err: %e)\n", backward_diff, fabs(exact_val - backward_diff));
    printf("Central Difference: %lf (Err: %e)\n", central_diff, fabs(exact_val - central_diff));
    return 0;
}

4.2 Trapezoidal Rule

Approximates the area under the curve using linear trapezoids over n subintervals of width h = (b - a) / n.

Trapezoidal Rule C Implementation:


#include <stdio.h>
#include <math.h>

// Function f(x) = 1 / (1 + x^2)
double f(double x) {
    return 1.0 / (1.0 + x*x);
}

double trapezoidal(double a, double b, int n) {
    double h = (b - a) / n;
    double sum = f(a) + f(b);
    
    for (int i = 1; i < n; i++) {
        sum += 2.0 * f(a + i*h);
    }
    return (h / 2.0) * sum;
}

int main() {
    double a = 0.0, b = 1.0;
    int n = 6;
    double integral = trapezoidal(a, b, n);
    printf("Trapezoidal Integration value: %lf (True value: %lf)\n", integral, atan(1.0));
    return 0;
}

4.3 Simpson's 1/3 and 3/8 Rules

Simpson's rules use higher-order polynomials to approximate the function over subintervals.

Simpson's Rules C Implementation:


#include <stdio.h>
#include <math.h>

double f(double x) {
    return 1.0 / (1.0 + x*x);
}

double simpson13(double a, double b, int n) {
    if (n % 2 != 0) {
        printf("Error: Simpson's 1/3 requires an even number of intervals.\n");
        return -1.0;
    }
    double h = (b - a) / n;
    double sum = f(a) + f(b);
    
    for (int i = 1; i < n; i++) {
        if (i % 2 == 0) {
            sum += 2.0 * f(a + i*h);
        } else {
            sum += 4.0 * f(a + i*h);
        }
    }
    return (h / 3.0) * sum;
}

double simpson38(double a, double b, int n) {
    if (n % 3 != 0) {
        printf("Error: Simpson's 3/8 requires intervals to be a multiple of 3.\n");
        return -1.0;
    }
    double h = (b - a) / n;
    double sum = f(a) + f(b);
    
    for (int i = 1; i < n; i++) {
        if (i % 3 == 0) {
            sum += 2.0 * f(a + i*h);
        } else {
            sum += 3.0 * f(a + i*h);
        }
    }
    return (3.0 * h / 8.0) * sum;
}

int main() {
    double a = 0.0, b = 1.0;
    printf("Simpson's 1/3: %lf\n", simpson13(a, b, 6));
    printf("Simpson's 3/8: %lf\n", simpson38(a, b, 6));
    return 0;
}

4.4 Romberg Integration

Romberg integration combines the Composite Trapezoidal Rule with Richardson Extrapolation to eliminate leading error terms, yielding highly accurate integration values.

Romberg Integration C Implementation:


#include <stdio.h>
#include <math.h>

double f(double x) {
    return 1.0 / (1.0 + x*x);
}

// Simple composite trapezoid helper
double trapezoid_helper(double a, double b, int n) {
    double h = (b - a) / n;
    double sum = f(a) + f(b);
    for (int i = 1; i < n; i++) {
        sum += 2.0 * f(a + i*h);
    }
    return (h / 2.0) * sum;
}

void romberg(double a, double b, int steps) {
    double R[10][10];
    
    // Step 1: Initialize first column with basic trapezoid values
    for (int i = 1; i <= steps; i++) {
        R[i][1] = trapezoid_helper(a, b, pow(2, i-1));
    }
    
    // Step 2: Compute extrapolated values
    for (int j = 2; j <= steps; j++) {
        for (int k = j; k <= steps; k++) {
            R[k][j] = R[k][j-1] + (R[k][j-1] - R[k-1][j-1]) / (pow(4, j-1) - 1.0);
        }
    }
    
    // Print Romberg Table
    printf("\nRomberg Integration Table:\n");
    for (int i = 1; i <= steps; i++) {
        for (int j = 1; j <= i; j++) {
            printf("%10.6f ", R[i][j]);
        }
        printf("\n");
    }
}

int main() {
    romberg(0.0, 1.0, 4);
    return 0;
}

4.5 Gaussian Quadrature (Two-Point Gauss Legendre)

Gaussian Quadrature evaluates the function at optimized, unequally spaced points to maximize accuracy.

Two-Point Gauss Legendre C Implementation:


#include <stdio.h>
#include <math.h>

// Function f(x) = x^2 + 1
double f_func(double x) {
    return x*x + 1.0;
}

double gauss_quadrature_2point(double a, double b) {
    // Gauss-Legendre points and weights for interval [-1, 1]
    double t0 = -1.0 / sqrt(3.0);
    double t1 = 1.0 / sqrt(3.0);
    double w0 = 1.0;
    double w1 = 1.0;
    
    // Transform variables from [a, b] to [-1, 1]
    double x0 = ((b - a) * t0 + (b + a)) / 2.0;
    double x1 = ((b - a) * t1 + (b + a)) / 2.0;
    double jacobian = (b - a) / 2.0;
    
    return jacobian * (w0 * f_func(x0) + w1 * f_func(x1));
}

int main() {
    double a = 0.0, b = 2.0;
    double integral = gauss_quadrature_2point(a, b);
    printf("Gauss Quadrature (2-point) over [%lf, %lf] = %lf\n", a, b, integral);
    return 0;
}
