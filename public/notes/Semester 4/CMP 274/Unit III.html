3.1 Linear Interpolation

Linear Interpolation connects two known points (x_0, y_0) and (x_1, y_1) with a straight line. The equation of the line is:


y = y_0 + [ (y_1 - y_0) * (x - x_0) ] / (x_1 - x_0)

3.2 Lagrange's Interpolation

Lagrange's Interpolation works for both equally and unequally spaced data points. Proving the uniqueness of this interpolating polynomial is standard in numerical analysis.

Lagrange's Interpolation C Implementation:


#include <stdio.h>

// Function to interpolate at target x using Lagrange's method
double lagrange_interpolate(double x[], double y[], int n, double target_x) {
    double result = 0.0;
    
    for (int i = 0; i < n; i++) {
        double term = y[i];
        for (int j = 0; j < n; j++) {
            if (j != i) {
                term = term * (target_x - x[j]) / (x[i] - x[j]);
            }
        }
        result += term;
    }
    return result;
}

int main() {
    double x[] = {1.0, 2.0, 5.0};
    double y[] = {3.0, 7.0, 19.0};
    int n = sizeof(x) / sizeof(x[0]);
    double target = 3.0;
    
    double interpolated_val = lagrange_interpolate(x, y, n, target);
    printf("Lagrange Interpolation at x = %lf is: %lf\n", target, interpolated_val);
    return 0;
}

3.3 Newton's Divided Difference Method

Constructs the interpolating polynomial for unequally spaced data points by evaluating divided differences.

Newton's Divided Difference C Implementation:


#include <stdio.h>

// Function to print Divided Difference Table and return interpolated value
double newton_divided_diff(double x[], double y[], int n, double target_x) {
    double dd[10][10];
    
    // Step 1: Initialize first column with y values
    for (int i = 0; i < n; i++) {
        dd[i][0] = y[i];
    }
    
    // Step 2: Build divided difference table
    for (int j = 1; j < n; j++) {
        for (int i = 0; i < n - j; i++) {
            dd[i][j] = (dd[i+1][j-1] - dd[i][j-1]) / (x[i+j] - x[i]);
        }
    }
    
    // Step 3: Print divided difference table
    printf("\nDivided Difference Table:\n");
    for (int i = 0; i < n; i++) {
        printf("%8.3f ", x[i]);
        for (int j = 0; j < n - i; j++) {
            printf("%8.3f ", dd[i][j]);
        }
        printf("\n");
    }
    
    // Step 4: Calculate interpolated value
    double sum = dd[0][0];
    double term = 1.0;
    for (int i = 1; i < n; i++) {
        term *= (target_x - x[i-1]);
        sum += term * dd[0][i];
    }
    return sum;
}

int main() {
    double x[] = {1.0, 2.0, 5.0};
    double y[] = {3.0, 7.0, 19.0};
    int n = sizeof(x) / sizeof(x[0]);
    double target = 3.0;
    
    double val = newton_divided_diff(x, y, n, target);
    printf("\nNewton Divided Difference value at x = %lf is: %lf\n", target, val);
    return 0;
}

3.4 Newton's Forward & Backward Difference Interpolation

Used specifically for equally spaced data points (where step size h = x_i+1 - x_i is constant).

Newton's Forward Difference C Implementation:


#include <stdio.h>

void newton_forward(double x[], double y[], int n, double target_x) {
    double diff[10][10];
    double h = x[1] - x[0];
    double u = (target_x - x[0]) / h;
    
    // Initialize first column of diff table
    for (int i = 0; i < n; i++) {
        diff[i][0] = y[i];
    }
    
    // Build difference table
    for (int j = 1; j < n; j++) {
        for (int i = 0; i < n - j; i++) {
            diff[i][j] = diff[i+1][j-1] - diff[i][j-1];
        }
    }
    
    // Calculate interpolated value
    double sum = diff[0][0];
    double term = 1.0;
    double u_term = 1.0;
    for (int i = 1; i < n; i++) {
        u_term *= (u - (i - 1));
        term *= i;
        sum += (u_term / term) * diff[0][i];
    }
    printf("Newton Forward value at x = %lf is: %lf\n", target_x, sum);
}

int main() {
    double x[] = {10.0, 20.0, 30.0, 40.0};
    double y[] = {1.1, 2.3, 3.9, 6.1};
    int n = sizeof(x) / sizeof(x[0]);
    newton_forward(x, y, n, 25.0);
    return 0;
}

3.6 Least Square Regression

Regression finds a curve that best fits the general trend of data points by minimizing the sum of the squared residuals (errors).

Fitting a Straight Line (y = a + bx) Normal Equations Derivation:

Let the residual error at point (x_i, y_i) be:
e_i = y_i - (a + b*x_i)

We minimize the sum of squared errors S = Σ (e_i)^2:
S = Σ [ y_i - (a + b*x_i) ]^2

To minimize S, we take the partial derivatives with respect to a and b and set them to zero:


∂S/∂a = -2 * Σ [ y_i - (a + b*x_i) ] = 0
       => Σ y_i - n*a - b * Σ x_i = 0
       => Σ y_i = n*a + b * Σ x_i    (Normal Equation 1)

∂S/∂b = -2 * Σ x_i * [ y_i - (a + b*x_i) ] = 0
       => Σ x_i*y_i - a * Σ x_i - b * Σ x_i^2 = 0
       => Σ x_i*y_i = a * Σ x_i + b * Σ x_i^2   (Normal Equation 2)

Linear Least Squares C Implementation:


#include <stdio.h>

void linear_regression(double x[], double y[], int n) {
    double sum_x = 0.0, sum_y = 0.0, sum_xx = 0.0, sum_xy = 0.0;
    
    for (int i = 0; i < n; i++) {
        sum_x += x[i];
        sum_y += y[i];
        sum_xx += x[i] * x[i];
        sum_xy += x[i] * y[i];
    }
    
    // Solve system of 2 equations using Cramer's rule:
    // n*a + sum_x*b = sum_y
    // sum_x*a + sum_xx*b = sum_xy
    
    double det = n * sum_xx - sum_x * sum_x;
    if (det == 0) {
        printf("Error: Determinant is zero. Cannot fit line.\n");
        return;
    }
    
    double a = (sum_y * sum_xx - sum_xy * sum_x) / det;
    double b = (n * sum_xy - sum_x * sum_y) / det;
    
    printf("Fitted Straight Line: y = %lf + %lfx\n", a, b);
}

int main() {
    double x[] = {1.0, 2.0, 3.0, 4.0};
    double y[] = {2.0, 5.0, 7.0, 10.0};
    int n = sizeof(x) / sizeof(x[0]);
    linear_regression(x, y, n);
    return 0;
}
